    // =========================================================================
    //  SECTION: _update OVERRIDE (ALL TRANSFER LOGIC)
    //
    //  OZ v5 architecture: ALL token movements (transfer, transferFrom, _mint,
    //  _burn) flow through _update(). One override handles everything safely.
    //
    //  [Bug-5 fix] No separate external transfer() / transferFrom() overrides.
    //              Those had wrong visibility (external vs public) and missing
    //              returns(bool). _update() is the correct OZ v5 pattern.
    //  [Bug-6 fix] No missing returns(bool) — _update has no return value.
    //  [Bug-7 fix] No unhandled return from super.transfer().
    //  [Bug-8 fix] Single clean override, no bypass risk.
    //  [Bug-4 fix] 90% rule uses integer math (* 90 / 100), not 0.9 float.
    // =========================================================================
    function _update(
        address from,
        address to,
        uint256 amount
    ) internal override(ERC20, ERC20Pausable) {

        // ── Pass mints and burns through without transfer restrictions ──────
        if (from == address(0) || to == address(0)) {
            super._update(from, to, amount);
            return;
        }

        // ── Blacklist ────────────────────────────────────────────────────────
        if (isBlacklisted[from]) revert DVX_Blacklisted(from);
        if (isBlacklisted[to])   revert DVX_Blacklisted(to);

        // ── Trading gate ─────────────────────────────────────────────────────
        if (!tradingEnabled) {
            require(
                isWhitelisted[from] || isWhitelisted[to],
                "DVX: Trading not enabled yet"
            );
        }

        // ── Detect buy/sell/transfer type ────────────────────────────────────
        bool fromAMM = isAMMPair[from];
        bool toAMM   = isAMMPair[to];
        bool isBuy   = fromAMM && !toAMM;
        bool isSell  = toAMM   && !fromAMM;
        bool excFees = isExcludedfromFees[from] || isExcludedfromFees[to];
        bool excLims = isExcludedFromLimits[from] || isExcludedFromLimits[to];

        // ── Vesting enforcement (owner non-sell transfers) ───────────────────
        // Checks before excLims — vesting is always enforced for team tokens
        if (from == owner() && !isSell && !ownerRenounced) {
            uint256 alreadyUsed = TEAM_TOKENS > balanceOf(owner())
                ? TEAM_TOKENS - balanceOf(owner()) : 0;
            uint256 newTotal = alreadyUsed + amount;
            uint256 vested   = _computeVested();
            if (newTotal > vested) {
                revert DVX_VestingExceeded(
                    vested > alreadyUsed ? vested - alreadyUsed : 0,
                    amount
                );
            }
            vestingClaimed = newTotal;
        }

        // ── Owner sell dump protection ────────────────────────────────────────
        if (from == owner() && isSell) {
            if (ownerLastSell != 0 && block.timestamp < ownerLastSell + OWNER_SELL_COOLDOWN) {
                revert DVX_DumpCooldown(ownerLastSell + OWNER_SELL_COOLDOWN);
            }
            uint256 ownerBal = balanceOf(from);
            uint256 cap      = ownerBal * OWNER_SELL_CAP_BPS / 10_000;
            if (amount > cap) revert DVX_DumpCapExceeded(cap, amount);
            ownerLastSell = block.timestamp;
            emit OwnerSell(from, amount);
        }

        // ── [Bug-4 fix] 90% max transfer rule (integer math only) ────────────
        // Cannot send more than 90% of your own balance per transaction.
        // Exempt: owner (has dump guard above), AMM pairs, excluded addresses.
        if (!excLims && !fromAMM && !toAMM && from != owner()) {
            uint256 bal = balanceOf(from);
            if (bal > 0) {
                uint256 cap90 = bal * TRANSFER_PCT_MAX / 100;
                if (amount > cap90) revert DVX_Transfer90PctExceeded(cap90, amount);
            }
        }

        // ── Anti-whale limits ─────────────────────────────────────────────────
        if (limitsEnabled && !excLims) {
            if (isBuy && buyCooldownSec > 0) {
                require(block.timestamp >= lastBuy[to] + buyCooldownSec, "DVX: Buy cooldown");
                lastBuy[to] = block.timestamp;
            }
            if (isSell && sellCooldownSec > 0) {
                require(block.timestamp >= lastSell[from] + sellCooldownSec, "DVX: Sell cooldown");
                lastSell[from] = block.timestamp;
            }
            if (amount > maxTx) revert DVX_MaxTxExceeded(maxTx, amount);
            if (!toAMM) {
                uint256 newBal = balanceOf(to) + amount;
                if (newBal > maxWallet) revert DVX_MaxWalletExceeded(maxWallet, newBal);
            }
        }

        // ── Launch phase expiry ───────────────────────────────────────────────
        bool launch = inLaunchPhase;
        if (launch && block.timestamp >= tradingEnabledAt + LAUNCH_WINDOW) {
            inLaunchPhase = false;
            launch        = false;
            emit LaunchOver(block.timestamp);
        }

        // ── Trigger auto-swap if enough fees accumulated ──────────────────────
        uint256 accumulated = buckLP + buckMkt + buckDev + buckStk;
        if (accumulated >= swapAt && !_swapping && isSell && !excFees) {
            _doSwap(accumulated > swapMax ? swapMax : accumulated);
        }

        // ── Compute and deduct fees ───────────────────────────────────────────
        uint256 feeTokens = 0;
        if (!excFees) {
            if (launch && (isBuy || isSell)) {
                // Flat 2% launch tax
                feeTokens = amount * LAUNCH_TAX_BPS / 10_000;
                if (feeTokens > 0) {
                    // Credit fee to contract using base ERC20 (no recursion)
                    ERC20._update(from, address(this), feeTokens);
                    unchecked {
                        buckLP  += feeTokens / 2;
                        buckMkt += feeTokens - feeTokens / 2;
                    }
                }
            } else if (isBuy) {
                feeTokens = _takeFee(
                    from, amount,
                    buyFeeTotal, buyFeeLP, buyFeeMkt, buyFeeDev, buyFeeStk
                );
            } else if (isSell) {
                feeTokens = _takeFee(
                    from, amount,
                    sellFeeTotal, sellFeeLP, sellFeeMkt, sellFeeDev, sellFeeStk
                );
            } else if (transferFee > 0) {
                feeTokens = amount * transferFee / 10_000;
                if (feeTokens > 0) {
                    ERC20._update(from, address(this), feeTokens);
                    unchecked { buckMkt += feeTokens; }
                }
            }
        }

        // ── Execute the net transfer ──────────────────────────────────────────
        super._update(from, to, amount - feeTokens);
    }

    // =========================================================================
    //  FEE COLLECTION
    //  Uses ERC20._update (base, no override) to avoid recursive fee loop.
    // =========================================================================
    function _takeFee(
        address from,
        uint256 amount,
        uint256 total,
        uint256 lpShare,
        uint256 mktShare,
        uint256 devShare,
        uint256 stkShare
    ) internal returns (uint256 fee) {
        if (total == 0) return 0;
        fee = amount * total / 10_000;
        if (fee == 0) return 0;

        uint256 lp  = fee * lpShare  / total;
        uint256 mkt = fee * mktShare / total;
        uint256 dev = fee * devShare  / total;
        uint256 stk = fee * stkShare  / total;
        uint256 brn = fee - lp - mkt - dev - stk;

        uint256 toContract = fee - brn;
        if (toContract > 0) {
            ERC20._update(from, address(this), toContract);
            unchecked {
                buckLP  += lp;
                buckMkt += mkt;
                buckDev += dev;
                buckStk += stk;
            }
        }
        if (brn > 0) {
            ERC20._update(from, address(0), brn);
        }
    }

    // =========================================================================
    //  VESTING (Team 10% — 3-year linear, 30-day cliff)
    // =========================================================================
    function _computeVested() internal view returns (uint256) {
        if (block.timestamp < vestingStart + VESTING_CLIFF) return 0;
        uint256 elapsed = block.timestamp - vestingStart;
        if (elapsed >= VESTING_DURATION) return TEAM_TOKENS;
        return TEAM_TOKENS * elapsed / VESTING_DURATION;
    }

    function vestedAmount() external view returns (uint256) {
        return _computeVested();
    }

    function claimableNow() external view returns (uint256) {
        uint256 v = _computeVested();
        return v > vestingClaimed ? v - vestingClaimed : 0;
    }

    // =========================================================================
    //  SWAP & LIQUIFY (auto-convert fee tokens → BNB → distribute)
    // =========================================================================
    function _doSwap(uint256 tokens) private swapLock {
        if (tokens == 0 || address(router) == address(0)) return;

        // LP reserve check — skip if pool is too shallow
        if (dexPair != address(0) && minPoolBNB > 0) {
            try IPancakePair02(dexPair).getReserves() returns (uint112 r0, uint112 r1, uint32) {
                uint256 bnbReserve;
                try IPancakePair02(dexPair).token0() returns (address t0) {
                    bnbReserve = (t0 == address(this)) ? uint256(r1) : uint256(r0);
                } catch {
                    bnbReserve = uint256(r0) < uint256(r1) ? uint256(r0) : uint256(r1);
                }
                if (bnbReserve < minPoolBNB) {
                    emit SwapSkipped_("Pool BNB below minimum");
                    return;
                }
            } catch {
                // Pool not ready — continue anyway (first liquidity event)
            }
        }

        uint256 halfLP  = buckLP / 2;
        uint256 toSwap  = tokens > halfLP ? tokens - halfLP : tokens;
        if (toSwap == 0) return;

        uint256 bnbBefore = address(this).balance;

        address[] memory path = new address[](2);
        path[0] = address(this);
        path[1] = router.WETH();

        router.swapExactTokensForETHSupportingFeeOnTransferTokens(
            toSwap, 0, path, address(this), block.timestamp + 300
        );

        uint256 bnbGained = address(this).balance - bnbBefore;
        if (bnbGained == 0) return;

        uint256 div = buckLP + buckMkt + buckDev + buckStk;
        if (div == 0) return;

        uint256 bnbLP  = halfLP > 0 ? bnbGained * halfLP / div : 0;
        uint256 bnbMkt = bnbGained * buckMkt / div;
        uint256 bnbDev = bnbGained * buckDev / div;

        if (halfLP > 0 && bnbLP > 0) {
            // Add LP: tokens go to dead address — permanently locked, rugpull proof
            router.addLiquidityETH{value: bnbLP}(
                address(this), halfLP, 0, 0, address(0), block.timestamp + 300
            );
            emit PoolSwapped(toSwap, bnbLP, halfLP);
        }
        if (bnbMkt > 0) _sendBNB(payable(marketingWallet), bnbMkt);
        if (bnbDev  > 0) _sendBNB(payable(devWallet), bnbDev);

        buckLP  = 0;
        buckMkt = 0;
        buckDev = 0;
        buckStk = 0;
    }

    function _sendBNB(address payable to, uint256 amount) private {
        if (amount == 0 || address(this).balance < amount) return;
        (bool ok,) = to.call{value: amount}("");
        require(ok, "DVX: BNB transfer failed");
    }

    // =========================================================================
    //  STAKING
    // =========================================================================
    function stake(uint256 amount, uint256 lockSeconds) external nonReentrant whenNotPaused {
        if (stakingPaused)                    revert DVX_Staking_Paused();
        if (stakes[msg.sender].amount > 0)    revert DVX_Staking_Already();
        if (amount == 0)                      revert DVX_ZeroAmount();
        if (lockSeconds > 365 days) lockSeconds = 365 days;

        // Transfer tokens from user to contract (bypass fee via ERC20._update)
        ERC20._update(msg.sender, address(this), amount);

        stakes[msg.sender] = StakeInfo(amount, block.timestamp, block.timestamp, lockSeconds);
        unchecked { totalStaked += amount; }
        emit Staked_(msg.sender, amount, lockSeconds / 1 days);
    }

    function pendingReward(address user) public view returns (uint256) {
        StakeInfo memory s = stakes[user];
        if (s.amount == 0) return 0;
        uint256 apy     = s.lockDuration >= BONUS_LOCK ? stakingAPY * 120 / 100 : stakingAPY;
        uint256 elapsed = block.timestamp - s.lastClaimed;
        return s.amount * apy * elapsed / (10_000 * 365 days);
    }

    function claimStakingReward() external nonReentrant whenNotPaused {
        StakeInfo storage s = stakes[msg.sender];
        if (s.amount == 0) revert DVX_Staking_NotFound();
        uint256 reward = pendingReward(msg.sender);
        if (reward == 0) revert DVX_Staking_NoReward();
        uint256 poolBal   = balanceOf(address(this));
        uint256 available = poolBal > totalStaked ? poolBal - totalStaked : 0;
        if (available < reward) revert DVX_Staking_PoolEmpty();
        s.lastClaimed = block.timestamp;
        ERC20._update(address(this), msg.sender, reward);
        emit RewardPaid(msg.sender, reward);
    }

    function unstake() external nonReentrant whenNotPaused {
        StakeInfo memory s = stakes[msg.sender];
        if (s.amount == 0) revert DVX_Staking_NotFound();
        if (block.timestamp < s.stakedAt + s.lockDuration) {
            revert DVX_Staking_Locked(s.stakedAt + s.lockDuration);
        }
        uint256 reward = pendingReward(msg.sender);
        delete stakes[msg.sender];
        unchecked { totalStaked -= s.amount; }

        uint256 available = balanceOf(address(this)) > totalStaked
            ? balanceOf(address(this)) - totalStaked : 0;
        if (reward > available) reward = available;

        ERC20._update(address(this), msg.sender, s.amount + reward);
        emit Unstaked_(msg.sender, s.amount, reward);
    }

    function emergencyUnstake() external nonReentrant {
        StakeInfo memory s = stakes[msg.sender];
        if (s.amount == 0) revert DVX_Staking_NotFound();
        delete stakes[msg.sender];
        unchecked { totalStaked -= s.amount; }
        ERC20._update(address(this), msg.sender, s.amount);
        emit EmergencyExit(msg.sender, s.amount);
    }

    // =========================================================================
    //  TIMELOCK HELPERS
    // =========================================================================
    function _queue(bytes32 id, uint256 delay, bytes memory data) internal {
        timelocks[id] = TimelockData(block.timestamp + delay, data, true);
        emit TimelockQ(id, block.timestamp + delay);
    }

    function _exec(bytes32 id) internal returns (bytes memory data) {
        TimelockData storage t = timelocks[id];
        if (!t.queued) revert DVX_TimelockNotQueued();
        if (block.timestamp < t.executeAt) revert DVX_TimelockPending(t.executeAt);
        data = t.payload;
        delete timelocks[id];
        emit TimelockExec(id);
    }

    function cancelTimelock(bytes32 id) external onlyOwnerOrGuardian {
        if (!timelocks[id].queued) revert DVX_TimelockNotQueued();
        delete timelocks[id];
        emit TimelockCancel(id);
    }

    // ─── Fee timelock functions ───────────────────────────────────────────────
    function queueBuyFee(uint256 lp, uint256 mkt, uint256 dev, uint256 brn, uint256 stk) external onlyOwner {
        uint256 t = lp + mkt + dev + brn + stk;
        require(t <= MAX_BUY_FEE, "DVX: Exceeds max buy fee");
        _queue(TL_BUY, TIMELOCK, abi.encode(lp, mkt, dev, brn, stk));
    }

    function execBuyFee() external onlyOwner {
        bytes memory d = _exec(TL_BUY);
        (uint256 lp, uint256 mkt, uint256 dev, uint256 brn, uint256 stk) =
            abi.decode(d, (uint256, uint256, uint256, uint256, uint256));
        buyFeeLP = lp; buyFeeMkt = mkt; buyFeeDev = dev; buyFeeBurn = brn; buyFeeStk = stk;
        buyFeeTotal = lp + mkt + dev + brn + stk;
        emit FeeChange(buyFeeTotal, sellFeeTotal);
    }

    function queueSellFee(uint256 lp, uint256 mkt, uint256 dev, uint256 brn, uint256 stk) external onlyOwner {
        uint256 t = lp + mkt + dev + brn + stk;
        require(t <= MAX_SELL_FEE, "DVX: Exceeds max sell fee");
        _queue(TL_SELL, TIMELOCK, abi.encode(lp, mkt, dev, brn, stk));
    }

    function execSellFee() external onlyOwner {
        bytes memory d = _exec(TL_SELL);
        (uint256 lp, uint256 mkt, uint256 dev, uint256 brn, uint256 stk) =
            abi.decode(d, (uint256, uint256, uint256, uint256, uint256));
        sellFeeLP = lp; sellFeeMkt = mkt; sellFeeDev = dev; sellFeeBurn = brn; sellFeeStk = stk;
        sellFeeTotal = lp + mkt + dev + brn + stk;
        emit FeeChange(buyFeeTotal, sellFeeTotal);
    }

    // =========================================================================
    //  ORACLE MARKET ADJUSTMENT
    // =========================================================================
    function onOracleUpdate(uint256 exportIndex) external {
        if (msg.sender != address(oracle)) revert DVX_OracleOnly();
        if (!oracleEnabled) return;
        emit OracleCall(exportIndex, block.timestamp);
        _applyMarket(exportIndex);
    }

    function _applyMarket(uint256 idx) internal {
        if (!oracleEnabled) return;
        if (address(oracle) == address(0)) return;
        if (block.timestamp < lastOracleTrigger + ORACLE_COOLDOWN) return;
        if (block.timestamp > oracle.lastUpdatedAt() + ORACLE_STALE) return;
        if (idx == 0) return;

        Market newM;
        uint256 newBuyLP;
        uint256 newBuyBrn;
        uint256 newSellBrn;

        if (idx > bullThreshold) {
            newM = Market.BULL;
            newBuyLP = 50; newBuyBrn = 200; newSellBrn = 300;
        } else if (idx < bearThreshold) {
            newM = Market.BEAR;
            newBuyLP = 200; newBuyBrn = 50; newSellBrn = 100;
        } else {
            newM = Market.NEUTRAL;
            newBuyLP = 100; newBuyBrn = 100; newSellBrn = 200;
        }

        if (newM == market) {
            lastOracleTrigger = block.timestamp; // update even if market unchanged
            return;
        }

        uint256 newBuyT  = newBuyLP + 200 + 100 + newBuyBrn + 100;
        uint256 newSellT = 100 + 300 + 100 + newSellBrn + 100;
        if (newBuyT > MAX_BUY_FEE || newSellT > MAX_SELL_FEE) return;

        // Clamp buy fee delta (max ±1% per update)
        if (newBuyT > buyFeeTotal + MAX_FEE_DELTA)         newBuyT = buyFeeTotal + MAX_FEE_DELTA;
        else if (buyFeeTotal > newBuyT + MAX_FEE_DELTA)    newBuyT = buyFeeTotal - MAX_FEE_DELTA;

        // Clamp sell fee delta (max ±1% per update) — [BUG FIX: was missing]
        if (newSellT > sellFeeTotal + MAX_FEE_DELTA)        newSellT = sellFeeTotal + MAX_FEE_DELTA;
        else if (sellFeeTotal > newSellT + MAX_FEE_DELTA)   newSellT = sellFeeTotal - MAX_FEE_DELTA;

        buyFeeLP    = newBuyLP;
        buyFeeBurn  = newBuyBrn;
        buyFeeTotal = newBuyT;
        // [BUG FIX: individual sell components were not updated — _takeFee uses them]
        sellFeeLP   = 100;
        sellFeeMkt  = 300;
        sellFeeDev  = 100;
        sellFeeStk  = 100;
        sellFeeBurn = newSellBrn;
        sellFeeTotal = newSellT;
        market      = newM;
        lastOracleTrigger = block.timestamp;

        emit MarketChange(newM, idx);
        emit FeeChange(buyFeeTotal, sellFeeTotal);
    }

    function _resetFees() internal {
        // [BUG FIX: all 10 fee variables now reset, not just 5]
        buyFeeLP    = 100; buyFeeMkt  = 200; buyFeeDev  = 100;
        buyFeeBurn  = 100; buyFeeStk  = 100; buyFeeTotal = 600;
        sellFeeLP   = 100; sellFeeMkt = 300; sellFeeDev  = 100;
        sellFeeBurn = 200; sellFeeStk = 100; sellFeeTotal = 800;
        market = Market.NEUTRAL;
    }

    function manualSwap() external onlyOwnerOrGuardian {
        require(guardianApproved, "DVX: Guardian must approve");
        uint256 bal = balanceOf(address(this));
        uint256 av  = bal > totalStaked ? bal - totalStaked : 0;
        require(av > 0, "DVX: Nothing to swap");
        _doSwap(av);
        guardianApproved = false;
    }

    function manualBNBSend(address payable to, uint256 amount) external onlyOwner {
        require(guardianApproved, "DVX: Guardian must approve");
        if (to == address(0)) revert DVX_ZeroAddress();
        _sendBNB(to, amount);
        guardianApproved = false;
    }

    function rescueToken(address token, address to) external onlyOwner {
        require(token != address(this), "DVX: Cannot rescue DVX");
        require(guardianApproved, "DVX: Guardian must approve");
        if (to == address(0)) revert DVX_ZeroAddress();
        uint256 bal = IERC20Rescue(token).balanceOf(address(this));
        require(bal > 0, "DVX: Zero balance");
        IERC20Rescue(token).transfer(to, bal);
        guardianApproved = false;
        emit ERC20Rescue_(token, bal);
    }

    function initiateOwnerTransfer(address candidate) external onlyOwner {
        if (candidate == address(0)) revert DVX_ZeroAddress();
        pendingOwner = candidate;
        emit OwnershipInit(candidate);
    }

    function acceptOwnership() external {
        if (msg.sender != pendingOwner) revert DVX_NotPendingOwner();
        address old  = owner();
        pendingOwner = address(0);
        _transferOwnership(msg.sender);
        _excludeAll(msg.sender);
        vestingStart    = block.timestamp;
        vestingClaimed  = 0;
        ownerLastSell   = 0;
        emit OwnershipTransferred(old, msg.sender);
    }

    function renounceOwnership() public override onlyOwner {
        uint256 bal = balanceOf(owner());
        if (bal > 0) ERC20._update(owner(), reserveWallet, bal);
        ownerRenounced = true;
        super.renounceOwnership();
        emit OwnerRenounced_(msg.sender);
    }

    function getVestingInfo() external view returns (
        uint256 total,
        uint256 cliffAt,
        uint256 endsAt,
        uint256 vestedSoFar,
        uint256 claimed,
        uint256 claimable
    ) {
        uint256 v = _computeVested();
        return (
            TEAM_TOKENS,
            vestingStart + VESTING_CLIFF,
            vestingStart + VESTING_DURATION,
            v,
            vestingClaimed,
            v > vestingClaimed ? v - vestingClaimed : 0
        );
    }

    function getStakeInfo(address user) external view returns (
        uint256 amount,
        uint256 stakedAt,
        uint256 lockDuration,
        uint256 unlockAt,
        uint256 reward,
        bool    locked
    ) {
        StakeInfo memory s = stakes[user];
        return (
            s.amount,
            s.stakedAt,
            s.lockDuration,
            s.stakedAt + s.lockDuration,
            pendingReward(user),
            block.timestamp < s.stakedAt + s.lockDuration
        );
    }

    function getPoolStatus() external view returns (
        uint256 poolBal,
        uint256 staked,
        uint256 available,
        bool    healthy
    ) {
        uint256 b = balanceOf(address(this));
        uint256 a = b > totalStaked ? b - totalStaked : 0;
        return (b, totalStaked, a, a > 0);
    }

    function getOwnerSellStatus() external view returns (
        uint256 ownerBal,
        uint256 maxSell,
        uint256 nextAt,
        bool    canSell
    ) {
        uint256 b    = balanceOf(owner());
        uint256 cap  = b * OWNER_SELL_CAP_BPS / 10_000;
        uint256 next = ownerLastSell + OWNER_SELL_COOLDOWN;
        return (b, cap, next, ownerLastSell == 0 || block.timestamp >= next);
    }

    function getTimelockInfo(bytes32 id) external view returns (
        bool    queued,
        uint256 executeAt,
        uint256 remainingSeconds
    ) {
        TimelockData memory t = timelocks[id];
        uint256 rem = (t.queued && block.timestamp < t.executeAt)
            ? t.executeAt - block.timestamp : 0;
        return (t.queued, t.executeAt, rem);
    }

    function circulatingSupply() external view returns (uint256) {
        return totalSupply() - balanceOf(address(0));
    }

    function totalBurned() external view returns (uint256) {
        uint256 ts = totalSupply();
        return TOTAL_SUPPLY > ts ? TOTAL_SUPPLY - ts : 0;
    }
